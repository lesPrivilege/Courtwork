import { mkdtemp, rm } from 'node:fs/promises';
import os from 'node:os';
import path from 'node:path';
import { RuntimeStore } from '/Users/lesprivilege/Projects/.worktrees/courtwork-runtime-selection-r0-20260923/app/server/store.mjs';
import { EXECUTOR_OPERATIONS, PI_EXECUTOR_ID, executorConfigurationRef } from '/Users/lesprivilege/Projects/.worktrees/courtwork-runtime-selection-r0-20260923/app/server/executor-choice-state.mjs';

const root = await mkdtemp(path.join(os.tmpdir(), 'cw-r1-luna-revision-reopen-'));
let store;
try {
  store = await new RuntimeStore({ dataDir: root }).open();
  const project = await store.createProject('Revision drift reopen');
  const make = revision => ({ adapterId: PI_EXECUTOR_ID, revision,
    configurationRef: executorConfigurationRef({ adapterId: PI_EXECUTOR_ID, revision, protocol: 'pi-session-manager' }),
    capabilities: Object.fromEntries(EXECUTOR_OPERATIONS.map(name => [name, { supported: true }])) });
  const firstDescriptor = make('pi-r1');
  const session = await store.createSession({ projectId: project.id, title: 'Pi', workspaceDir: root, executorDescriptor: firstDescriptor });
  const makeRun = (commandId, descriptor) => ({ sessionId: session.id, input: 'x', adapterId: descriptor.adapterId,
    provider: { provider: 'fake-openai-loopback', model: 'fake-model', api: 'openai-completions', realProvider: false },
    extension: null, commandId, workspaceHostSession: null, credentialGeneration: 0, executorDescriptor: descriptor,
    expectedExecutorChoice: { revision: store.getSession(session.id).executorChoice.revision, adapterId: PI_EXECUTOR_ID } });
  const first = await store.createRun(makeRun('rev-1', firstDescriptor));
  await store.updateRun(first.run.id, { status: 'completed', admissionOpen: false });
  // Deliberately stale/reused fingerprint with a different revision tests whether
  // the Store enforces same-Session bound revision, independently of its factory.
  const drifted = { ...make('pi-r2'), configurationRef: firstDescriptor.configurationRef };
  await store.createRun(makeRun('rev-2', drifted));
  await store.close(); store = null;
  store = await new RuntimeStore({ dataDir: root }).open();
  console.log('COUNTEREXAMPLE persisted and reopened: same Session/ref contains adapter revisions ' +
    [...new Set(store.listRuns(session.id).map(run => run.executorBinding.revision))].join(', '));
} finally {
  await store?.close();
  await rm(root, { recursive: true, force: true });
}
