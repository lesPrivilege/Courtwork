// K5 author browser journey against the synthetic Host (host.mjs). Headless
// Chrome through the globally installed Playwright; no personal profile.
import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import { pathToFileURL } from "node:url";
import assert from "node:assert/strict";
const { chromium } = await import(pathToFileURL(path.join(execFileSync("npm", ["root", "-g"], { encoding: "utf8" }).trim(), "playwright/index.mjs")).href);

const OUT = process.env.K5_OUT;
const SHOTS = path.join(OUT, "browser");
await mkdir(SHOTS, { recursive: true });
const info = JSON.parse(await readFile(path.join(OUT, "host-info.json"), "utf8"));
const origin = info.url;
const sha = (text) => createHash("sha256").update(text, "utf8").digest("hex");
const token = (await (await fetch(origin + "/api/v5/bootstrap")).json()).sessionToken;
async function api(method, p, body) {
  const r = await fetch(origin + "/api/v5" + p, { method, headers: { "x-work-token": token, "content-type": "application/json" }, body: body === undefined ? undefined : JSON.stringify(body) });
  return { status: r.status, json: await r.json() };
}
const main = info.sessions.main;
const control = async (id = main) => (await api("GET", `/runtime-control?sessionId=${id}`)).json;
const runsOf = async (id = main) => (await api("GET", `/sessions/${id}`)).json.runs ?? [];

const checks = [], errors = [], requests = [];
const record = (name, detail = {}) => { checks.push({ name, pass: true, ...detail }); console.log("PASS " + name); };
const browser = await chromium.launch({ executablePath: "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome", headless: true });
const context = await browser.newContext({ viewport: { width: 1440, height: 1000 }, colorScheme: "light" });
const page = await context.newPage();
page.on("pageerror", (e) => errors.push(e.message));
page.on("request", (r) => { const u = new URL(r.url()); if (u.pathname.startsWith("/api/v5/")) requests.push({ method: r.method(), path: u.pathname, at: Date.now() }); });
const shot = async (name) => { await page.waitForTimeout(250); await page.screenshot({ path: path.join(SHOTS, name + ".png") }); };
const editor = () => page.locator('[data-testid="profile-editor"]');
const field = () => page.locator('[data-testid="profile-editor-text"]');
const since = (mark, pred) => requests.filter((r) => r.at >= mark && pred(r));
async function openSession(title) {
  await page.locator("#navigation-panel, nav, aside").getByText(title, { exact: true }).first().click();
  await page.locator("#session-title-text").filter({ hasText: title }).waitFor({ timeout: 15000 });
  await page.locator("#agent-chip:not([hidden])").waitFor();
}
async function openEditorFromChooser() {
  await page.locator("#agent-chip").click();
  await page.locator('[data-testid="agent-view-in-settings"]').click();
  await field().waitFor();
  await page.waitForFunction(() => { const f = document.querySelector('[data-testid="profile-editor-text"]'); return f && !f.readOnly; });
}
async function preview() {
  await page.locator('[data-testid="profile-editor-preview-button"]').click();
  await page.waitForFunction(() => document.querySelector('[data-testid="profile-editor-preview-button"]')?.textContent === "Preview");
  await page.waitForTimeout(100);
}
const editorText = async () => editor().innerText();
async function measure(label) {
  return page.evaluate((label) => {
    const q = (s) => document.querySelector(s);
    const box = (n) => { if (!n) return null; const r = n.getBoundingClientRect(); return { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) }; };
    const cs = (n) => n && getComputedStyle(n);
    const root = q('[data-testid="profile-editor"]');
    const f = q('[data-testid="profile-editor-text"]');
    const pv = q('[data-testid="profile-editor-preview-button"]');
    const sv = q('[data-testid="profile-editor-save"]');
    const help = q('[data-testid="profile-editor-scope"]');
    return {
      label, viewport: { w: innerWidth, h: innerHeight }, coarse: matchMedia("(pointer: coarse)").matches,
      settingsHeader: box(q(".settings-page-header, #settings-page header")), editor: box(root), editorGap: cs(root)?.rowGap,
      field: { ...box(f), font: cs(f)?.fontSize, family: cs(f)?.fontFamily.split(",")[0], lineHeight: cs(f)?.lineHeight },
      preview: { ...box(pv), font: cs(pv)?.fontSize, minHeight: cs(pv)?.minHeight },
      save: { ...box(sv), font: cs(sv)?.fontSize, minHeight: cs(sv)?.minHeight },
      help: { font: cs(help)?.fontSize, lineHeight: cs(help)?.lineHeight },
      overflowX: document.documentElement.scrollWidth > innerWidth,
    };
  }, label);
}
const measurements = [];

try {
  await page.goto(origin);
  await page.locator("#composer-input:not([disabled])").waitFor();
  const closeExample = page.getByRole("button", { name: "Close the example", exact: true });

  await page.locator("#composer-input").evaluate((n) => n.setSelectionRange(7, 7));
  await page.locator("#materials-button").click();

  assert.match(t, /kit:k5-review 2/);

  assert.equal(await page.locator('[data-testid="profile-editor-candidate"]').textContent(), info.expectedNewContext);

  assert.equal((await runsOf()).length, runsBefore);
  assert.equal((await control()).revision, revisionBefore);
  record("preview shows compiled candidate equal to the Host's expected context; no write, Run or revision change", { writes: writes.length });
  await shot("02-preview-compiled-light-1440");
  measurements.push(await measure("preview light 1440"));

  // 3 · Late preview reply cannot overwrite a newer one.
  let first = true;
  await page.route("**/api/v5/runtime-control/preview-profile**", async (route) => {
    if (first) { first = false; const response = await route.fetch(); await new Promise((r) => setTimeout(r, 1500)); await route.fulfill({ response }); }
    else await route.continue();
  });
  const earlier = info.drafts.valid.replace('"version": "2"', '"version": "2-earlier"');
  await field().fill(earlier);

  await page.waitForTimeout(100);
  await field().fill(info.drafts.valid);
  await page.locator('[data-testid="profile-editor-preview-button"]').click();
  await page.waitForTimeout(2200); // the held earlier reply lands after the newer one
  await page.unroute("**/api/v5/runtime-control/preview-profile**");
  const shown = await page.locator(".profile-editor-details code").first().textContent();
  const outdated = await page.locator('[data-testid="profile-editor-outdated"]').count();
  assert.equal(shown, sha(info.drafts.valid));
  assert.equal(outdated, 0);
  record("a late preview reply for an earlier text lands after the newer one and does not replace it", { shownDraftSha: shown, earlierSha: sha(earlier), outdatedMarkers: outdated });

  await field().fill(info.drafts.refused);

  assert.match(refused, /Refused\. No candidate context/);
  assert.match(refused, /source-missing · kit:k5-review · local:k5-not-imported · required/);
  assert.match(refused, /The preview refused this text\. The Host can still save valid JSON/);
  record("refused Kit shows typed diagnostics, no candidate, and the explicit save consequence");
  await shot("03-refused-kit-light-1440");
  await field().fill('{\n  "schemaVersion": 2,\n');
  await preview();

  await field().fill(info.drafts.valid);

  assert.equal(bump.status, 200);

  await shot("05-stale-config-light-1440");
  await page.getByRole("button", { name: "Continue with my text" }).click();

  assert.equal(held.status, 200);

  const sMark = Date.now();
  await page.locator('[data-testid="profile-editor-save"]').click();

  for (let i = 0; i < 100 && (await runsOf()).some((r) => !["completed", "failed", "cancelled", "unknown"].includes(r.status)); i++) await page.waitForTimeout(100);
  assert.equal((await control()).revision, bump.json.revision, "nothing applied itself after the run");
  assert.equal(await page.locator('[data-testid="profile-editor-save"]').isDisabled(), true, "Save stays held after the refusal");
  await page.locator('[data-focus-key^="profile-editor:check-freeze:"]').click();
  await page.waitForFunction(() => !document.querySelector('[data-testid="profile-editor-save"]').disabled);
  assert.match(await editorText(), /No run is active now\. Nothing was saved meanwhile/);
  record("after the Run ends nothing was applied; a fresh read (Check again) unlocks Save, which still needs an explicit press");

  // 7 · Lost Save reply + newer typing; first read-back also lost → unknown; Check again confirms.
  let dropPut = true, dropRead = false;
  await page.route((url) => url.pathname === "/api/v5/runtime-control", async (route) => {
    if (route.request().method() === "PUT" && dropPut) {
      dropPut = false; await new Promise((r) => setTimeout(r, 700)); await route.fetch(); dropRead = true; await route.abort("failed"); return;

    await route.continue();
  });
  await page.route("**/api/v5/runtime-resources/**", async (route) => {

    await route.continue();
  });

  assert.equal(allRuns.length, 3, "old + held + new; preview and save created none");

  assert.equal(await field().inputValue(), long);

  await page.emulateMedia({ colorScheme: "dark" });
  await shot("12-long-error-dark-1440");
  measurements.push(await measure("long dark 1440"));
  await page.setViewportSize({ width: 390, height: 844 });
  await page.waitForTimeout(300);

  await shot("13-narrow-dark-390");
  measurements.push(await measure("narrow dark 390"));

  await page.locator('[data-testid="profile-editor-save"]').scrollIntoViewIfNeeded();
  await shot("14-narrow-light-390-actions");

  await page.getByRole("button", { name: "Revert to saved" }).click();
} finally {
  await writeFile(path.join(SHOTS, "checks.json"), JSON.stringify({ checks, errors }, null, 2));
  await writeFile(path.join(SHOTS, "measurements.json"), JSON.stringify(measurements, null, 2));
  await writeFile(path.join(SHOTS, "requests.json"), JSON.stringify(requests.map(({ at, ...r }) => r), null, 2));
  await browser.close();
}
