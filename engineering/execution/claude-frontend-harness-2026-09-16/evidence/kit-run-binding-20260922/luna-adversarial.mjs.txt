// Probe 2: multi-Kit declaration/evidence permutations normalize to one plan identity.
{
  const input = fixture('combined');
  const base = { binding: clone(input.binding), kits: clone(input.kits), runtime: clone(input.runtime), compatibilityEvidence: [], budget: clone(input.budget) };
  const forward = planKitContext(base);
  const reversed = planKitContext({ ...clone(base), kits: clone(base.kits).reverse() });
  assert.equal(forward.status, 'compiled'); assert.equal(reversed.status, 'compiled');
  assert.equal(forward.planSha256, reversed.planSha256);
  assert.equal(forward.candidate.sha256, reversed.candidate.sha256);
  assert.deepEqual(forward.kits, reversed.kits);
  pass('multi-kit permutation', `plan=${forward.planSha256} candidate=${forward.candidate.sha256}`);
}

// Probe 3: selection/scope gate is checked before any retention write; accepted Session scope writes exactly two blobs.
{
  const input = fixture('combined');
  const session = { id: 'selected-session', scope: 'project', extensionBinding: null };
  const binding = clone(input.binding);
  binding.composition = { ...binding.composition, id: 'agent:k3', version: 'v2', hash: 'a'.repeat(64), status: 'compatible', schemaVersion: 2, kits: clone(input.kits), selectionScope: { type: 'user', id: 'local' } };
  let writes = 0;
  const history = { save: async () => { writes++; }, read: async () => { throw new Error('not used'); } };
  await assert.rejects(() => retainKitContext({ binding, session, spark: false, adapter: PI, history }), error => error.code === 'kit_scope_unsupported');
  assert.equal(writes, 0);
  binding.composition.selectionScope = { type: 'session', id: session.id };
  const summary = await retainKitContext({ binding, session, spark: false, adapter: PI, history });
  assert.equal(writes, 2); assert.equal(summary.adapter.id, PI.id);
  pass('selection guard', 'user scope refused with zero saves; selected Session scope retained exactly two payloads');
}

// Probe 4: retained plan/context disagreement fails closed and never falls back to mutable binding text.
{
  const input = fixture('combined');
  const session = { id: 'read-session', scope: 'project', extensionBinding: null };
  const binding = clone(input.binding);
  binding.composition = { ...binding.composition, id: 'agent:k3', version: 'v2', hash: 'a'.repeat(64), status: 'compatible', schemaVersion: 2, kits: clone(input.kits), selectionScope: { type: 'session', id: session.id } };
  const saved = new Map();
  const history = { save: async (_id, bytes, digest) => saved.set(digest, Buffer.from(bytes)), read: async (_id, digest) => Buffer.from(saved.get(digest)) };
  const summary = await retainKitContext({ binding, session, spark: false, adapter: PI, history });
  const good = await readKitContext(history, session.id, summary);
  assert.equal(good.text, good.plan.candidate.text);
  saved.set(summary.contextPayload.sha256, Buffer.from('mutable-current-context'));
  await assert.rejects(() => readKitContext(history, session.id, summary), error => error.code === 'kit_payload_invalid');
  pass('retention consistency', 'tampered retained context rejected with kit_payload_invalid and no fallback read');
}
